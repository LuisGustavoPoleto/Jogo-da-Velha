public static void main(String args[]) {
        Scanner ler = new Scanner(System.in);
        Random sorte = new Random();      
            int[][] bulero = new int[4][4];
            int jogador = 1, vencedor = 0, jogadas = 0;
            int linha = 0, coluna = 0;

        while (vencedor == 0 && jogadas < 9) {
            System.out.println("Tabuleiro Atual");
            for (int i = 1; i < 4; i++) {
                for (int j = 1; j < 4; j++) {
                    System.out.print(bulero[i][j] + " ");
            }
            System.out.println();
            }

            if (jogador == 1) {
                System.out.println("Sua vez!\n");
                System.out.println("Escolha uma linha (1 a 3) e uma coluna (1 a 3):");
                linha = ler.nextInt();
                coluna = ler.nextInt();
                if (linha < 1 || linha > 3 || coluna < 1 || coluna > 3) {
                    System.out.println("Posicao fora do tabuleiro!");
                    linha = 0;
                    coluna = 0;
                } else if (bulero[linha][coluna] != 0) {
                    System.out.println("Essa posicao ja foi marcada!");
                    linha = 0;
                    coluna = 0;
                }
            } else {
                System.out.println("Vez do gpt...");
               
                while (linha == 0 || coluna == 0) {
                    int linhaBot = sorte.nextInt(3) + 1;
                    int colunaBot = sorte.nextInt(3) + 1;

                    if (bulero[linhaBot][colunaBot] == 0) {
                        linha = linhaBot;
                        coluna = colunaBot;
                }
                }
                System.out.println("O gpt jogou na linha " + linha + " e coluna " + coluna);
            }

            if (linha != 0) {
                bulero[linha][coluna] = jogador;
                jogadas++;

                for (int i = 1; i < 4; i++) {
                    if (bulero[i][1] == jogador && bulero[i][2] == jogador && bulero[i][3] == jogador) {
                        vencedor = jogador;
                }
                }

                for (int j = 1; j < 4; j++) {
                    if (bulero[1][j] == jogador && bulero[2][j] == jogador && bulero[3][j] == jogador) {
                        vencedor = jogador;
                }
                }

                if (bulero[1][1] == jogador && bulero[2][2] == jogador && bulero[3][3] == jogador) {
                    vencedor = jogador;
                }

                if (bulero[1][3] == jogador && bulero[2][2] == jogador && bulero[3][1] == jogador) {
                    vencedor = jogador;
                }

                if (vencedor == 0) {
                    if (jogador == 1) {
                        jogador = 2;
                } else {
                        jogador = 1;
                }
                }
        }
        }

        System.out.println("Tabuleiro Final");
        for (int i = 1; i < 4; i++) {
            for (int j = 1; j < 4; j++) {
                System.out.print(bulero[i][j] + " ");
        }
        System.out.println();
        }
        if (vencedor == 1) {
            System.out.println("Você venceu o jogo!");
        } else if (vencedor == 2) {
            System.out.println("O gpt venceu o jogo!");
        } else {
            System.out.println("O jogo empatou noob!");
        }

}
